98 PadeLegendreSolver(const ReferenceElement& ref, Index L, Index M,
99 Real fallback_tol = Real(100) * std::numeric_limits<Real>::epsilon())
100 : ref_(ref), L_(L), M_(M), N_(ref.order()), Np_(ref.numDOF()),
101 fallback_tol_(fallback_tol)
102 {
103 if (L < 0 || M < 0)
104 throw std::invalid_argument("PadeLegendreSolver: L and M must be >= 0");
105 if (L + M > N_)
106 throw std::invalid_argument("PadeLegendreSolver: L + M must be <= N");
107
108 // Need 2*nq-1 >= k_max + m_max + N = N + M + N = 2N + M
109 // nq = N + M + 1 gives 2(N+M+1)-1 = 2N+2M+1 >= 2N+M ✓
110 const Index nq = N_ + M_ + 1;
111 }

125 std::vector<Real> modalCoeffs(const Real* nodal_vals) const
126 {
127 std::vector<Real> c(Np_);
128 for (Index n = 0; n < Np_; ++n) {
129 Real s = Real(0);
130 for (Index i = 0; i < Np_; ++i)
131 s += vinv_(n, i) * nodal_vals[i];
132 // vinv_ applied to nodal gives normalized coefficient c_hat_n = s
133 // unnormalize: c_n = c_hat_n * sqrt((2n+1)/2)
134 c[n] = s * std::sqrt(Real(2 * n + 1) / Real(2));
135 }
136 return c;
137 }

148 PadeApproximant<Real, Index> buildApproximant(const std::vector<Real>& c) const
149 {
150 PadeApproximant<Real, Index> approx;
151 approx.L = L_;
152 approx.M = M_;
153 approx.modal_coeffs = c;
154 approx.p_coeffs.assign(L_ + 1, Real(0));
155 approx.q_coeffs.assign(M_ + 1, Real(0));
156 approx.q_coeffs[0] = Real(1); // normalization
157
158 // B(k,m) = integral_{-1}^1 P_k(x) P_m(x) u_h(x) dx
159 const EigenMat B = buildBMatrix_(c);
160
161 if (M_ == 0) {
162 // Denominator is trivially 1; numerator equals polynomial truncation
163 for (Index k = 0; k <= L_; ++k)
164 approx.p_coeffs[k] = static_cast<Real>(
165 double(2 * k + 1) / 2.0 * B(k, 0));
166 approx.valid = true;
167 return approx;
168 }
169
170 // ── Denominator system: A q_vec = -b ────────────────────────────────
171 // Equations for k = L+1 .. N → (N-L) equations, M unknowns
172 const Index nRows = N_ - L_;
173 EigenMat A(nRows, M_);
174 EigenVec b(nRows);
175
176 for (Index k = L_ + 1; k <= N_; ++k) {
177 const Index row = k - L_ - 1;
178 b(row) = B(k, 0);
179 for (Index m = 1; m <= M_; ++m)
180 A(row, m - 1) = B(k, m);
181 }
182
183 // Least-squares solve (handles square and overdetermined systems)
184 const EigenVec q_vec = A.colPivHouseholderQr().solve(-b);
185
186 for (Index m = 1; m <= M_; ++m)
187 approx.q_coeffs[m] = static_cast<Real>(q_vec(m - 1));
188
189 // ── Numerator: p_k = (2k+1)/2 * sum_{m=0}^M q_m B(k,m) ─────────────
190 for (Index k = 0; k <= L_; ++k) {
191 double s = 0.0;
192 for (Index m = 0; m <= M_; ++m)
193 s += static_cast<double>(approx.q_coeffs[m]) * B(k, m);
194 approx.p_coeffs[k] = static_cast<Real>(double(2 * k + 1) / 2.0 * s);
195 }
196
197 approx.valid = true;
198 return approx;
199 }

211 Real evaluate(const PadeApproximant<Real, Index>& approx, Real x) const
212 {
213 Real P = Real(0);
214 for (Index l = 0; l <= approx.L; ++l)
215 P += approx.p_coeffs[l] * ReferenceElement::legendreP(l, x);
216
217 Real Q = Real(0);
218 for (Index m = 0; m <= approx.M; ++m)
219 Q += approx.q_coeffs[m] * ReferenceElement::legendreP(m, x);
220
221 if (std::abs(Q) < fallback_tol_) {
222 // Polynomial fallback: evaluate the original DG approximation
223 Real poly = Real(0);
224 for (Index n = 0; n < static_cast<Index>(approx.modal_coeffs.size()); ++n)
225 poly += approx.modal_coeffs[n] * ReferenceElement::legendreP(n, x);
226 return poly;
227 }
228
229 return P / Q;
230 }

242 std::vector<Real> reconstruct(const Real* nodal_vals,
243 const std::vector<Real>& output_nodes) const
244 {
245 const auto c = modalCoeffs(nodal_vals);
246 const auto approx = buildApproximant(c);
247 std::vector<Real> out(output_nodes.size());
248 for (std::size_t i = 0; i < output_nodes.size(); ++i)
249 out[i] = evaluate(approx, output_nodes[i]);
250 return out;
251 }

263 Field reconstruct(const Field& u) const
264 {
265 Field result(u.numElements(), Np_);
266
267 // Collect GLL node positions once
268 std::vector<Real> gll_nodes(Np_);
269 for (Index i = 0; i < Np_; ++i)
270 gll_nodes[i] = ref_.nodes()[i];
271
272 for (Index k = 0; k < u.numElements(); ++k) {
273 const Real* uk = u.elementPtr(k);
274 const auto vals = reconstruct(uk, gll_nodes);
275 Real* rk = result.elementPtr(k);
276 for (Index i = 0; i < Np_; ++i)
277 rk[i] = vals[i];
278 }
279 return result;
280 }

295 EigenMat buildBMatrix_(const std::vector<Real>& c) const
296 {
297 const Index nq = static_cast<Index>(ref_.numDOF());
298 EigenMat B = EigenMat::Zero(N_ + 1, M_ + 1);
299
300 for (Index q = 0; q < nq; ++q) {
301 const Real xq = ref_.nodes()[q];
302 const Real wq = ref_.weights()[q];
303
304 // Evaluate u_h at quadrature point
305 Real uh = Real(0);
306 for (Index n = 0; n <= N_; ++n)
307 uh += c[n] * ReferenceElement::legendreP(n, xq);
308
309 // Accumulate B(k, m) += w_q * P_k(x_q) * P_m(x_q) * u_h(x_q)
310 for (Index k = 0; k <= N_; ++k) {
311 const double Pk = static_cast<double>(ReferenceElement::legendreP(k, xq));
312 for (Index m = 0; m <= M_; ++m) {
313 const double Pm = static_cast<double>(ReferenceElement::legendreP(m, xq));
314 B(k, m) += static_cast<double>(wq) * Pk * Pm
315 * static_cast<double>(uh);
316 }
317 }
318 }
319 return B;
320 }
