25// ─────────────────────────────────────────────────────────────────────────────
26// writeToVTK
27//
28// Writes the DG solution to a legacy ASCII VTK file that Paraview can open.
29//
30// Strategy: the DG solution lives on sub-element nodes, not on the mesh
31// vertices. We therefore write a *refined* point cloud: for each element k
32// we emit its Np physical node positions as VTK points, and connect each
33// group of Np points into a VTK_POLY_LINE cell. This gives Paraview a
34// piecewise-polynomial picture of u inside every element, with the expected
35// discontinuities visible at element boundaries.
36//
37// Output layout:
38// - K * Np points (physical DG node positions)
39// - K cells (one VTK_POLY_LINE per element, with Np vertices)
40// - 1 point data array named after `fieldName`
41//
42// Usage:
43// DG::writeToVTK(mesh, ref, u, "output.vtk", "u");
44// ─────────────────────────────────────────────────────────────────────────────
45template
46<
47 class Real = double,
48 class Device = TNL::Devices::Host,
49 class Index = int
50>
59void writeToVTK(const Mesh<Real, Device, Index>& mesh,
60 const ReferenceElement<Real, Index>& ref,
61 const FieldVector<Real, Device, Index>& u,
62 const std::string& filename,
63 const std::string& fieldName = "u",
64 Real t = Real(0))
65{
66 const Index K = mesh.numElements();
67 const Index Np = ref.numDOF();
68 const Index totalPts = K * Np;
69
70 std::ofstream f(filename);
71 if (!f)
72 throw std::runtime_error("writeToVTK: cannot open '" + filename + "'");
73
74 f << std::scientific << std::setprecision(10);
75
76 // ── VTK header ──────────────────────────────────────────────────────────
77 f << "# vtk DataFile Version 3.0\n";
78 f << "DG1D solution: " << fieldName << " t=" << t << "\n";
79 f << "ASCII\n";
80 f << "DATASET UNSTRUCTURED_GRID\n";
81
82 // ── Points ──────────────────────────────────────────────────────────────
83 // VTK always needs 3D coordinates; pad y=0, z=0 for 1D.
84 f << "POINTS " << totalPts << " double\n";
85 for (Index k = 0; k < K; ++k)
86 {
87 const Real xL = mesh.leftVertex(k);
88 const Real h = mesh.elementSize(k);
89 for (Index i = 0; i < Np; ++i)
90 {
91 // map reference node r_i in [-1,1] to physical coordinate
92 const Real r = ref.nodes()[i];
93 const Real x = xL + (r + Real(1)) * Real(0.5) * h;
94 f << x << " 0.0 0.0\n";
95 }
96 }
97
98 // ── Cells ────────────────────────────────────────────────────────────────
99 // Each cell is a VTK_POLY_LINE with Np vertices.
100 // Cell list entry: [count, v0, v1, ..., v_{Np-1}]
101 const Index cellListSize = K * (Np + 1); // Np vertices + 1 count per cell
102 f << "\nCELLS " << K << " " << cellListSize << "\n";
103 for (Index k = 0; k < K; ++k)
104 {
105 f << Np;
106 for (Index i = 0; i < Np; ++i)
107 {
108 f << " " << k * Np + i;
109 }
110 f << "\n";
111 }
112
113 // VTK_POLY_LINE = cell type 4
114 f << "\nCELL_TYPES " << K << "\n";
115 for (Index k = 0; k < K; ++k)
116 {
117 f << "4\n";
118 }
119
120 // ── Point data ───────────────────────────────────────────────────────────
121 f << "\nPOINT_DATA " << totalPts << "\n";
122 f << "SCALARS " << fieldName << " double 1\n";
123 f << "LOOKUP_TABLE default\n";
124 for (Index k = 0; k < K; ++k)
125 {
126 const Real* uk = u.elementPtr(k);
127 for (Index i = 0; i < Np; ++i)
128 {
129 f << uk[i] << "\n";
130 }
131 }
132}

137// Convenience wrapper: writes a numbered sequence of VTK files suitable for

171{

178// ─────────────────────────────────────────────────────────────────────────────
179// writePadeVTK
180//
181// Writes the Padé–Legendre reconstruction to a legacy ASCII VTK file.
182//
183// Strategy: for each element k, `refineFactor * Np` equidistant reference
184// points are generated on [-1,1], the Padé approximant is evaluated there,
185// and the points are connected into a VTK_POLY_LINE cell. This gives a
186// smooth, high-resolution visualization of the rational approximant,
187// independent of the (irregularly-spaced) GLL nodes used by the DG solver.
188//
189// Usage:
190// DG::writePadeVTK(mesh, ref, u, pade_solver, "output/pade.vtk", "u_pade", t, 4);
191//
192// Parameters:
193// mesh — the DG mesh
194// ref — reference element (provides Np)
195// u — DG solution FieldVector (nodal values at GLL nodes)
196// solver — a constructed PadeLegendreSolver (same L,M,N as desired)
197// filename — output file path
198// fieldName — scalar field name written to the VTK header
199// t — simulation time (written to the VTK comment line)
200// refineFactor — number of equidistant sub-points per GLL node (default 4)
201// ─────────────────────────────────────────────────────────────────────────────
202template
203<
204 class Real = double,
205 class Device = TNL::Devices::Host,
206 class Index = int
207>
225void writePadeVTK(const Mesh<Real, Device, Index>& mesh,
226 const ReferenceElement<Real, Index>& ref,
227 const FieldVector<Real, Device, Index>& u,
228 const PadeLegendreSolver<Real, Index>& solver,
229 const std::string& filename,
230 const std::string& fieldName = "u_pade",
231 Real t = Real(0),
232 Index refineFactor = Index(4))
233{
234 if (refineFactor < 1)
235 throw std::invalid_argument("writePadeVTK: refineFactor must be >= 1");
236
237 const Index K = mesh.numElements();
238 const Index Np = ref.numDOF();
239 const Index nFine = refineFactor * Np; // fine points per element
240 const Index totalPts = K * nFine;
241
242 // Build equidistant reference coordinates once
243 std::vector<Real> ref_pts(nFine);
244 for (Index i = 0; i < nFine; ++i)
245 ref_pts[i] = Real(-1) + Real(2) * i / Real(nFine - 1);
246
247 std::ofstream f(filename);
248 if (!f)
249 throw std::runtime_error("writePadeVTK: cannot open '" + filename + "'");
250
251 f << std::scientific << std::setprecision(10);
252
253 // ── VTK header ────────────────────────────────────────────────────────────
254 f << "# vtk DataFile Version 3.0\n";
255 f << "DG1D Pade-Legendre reconstruction: " << fieldName << " t=" << t << "\n";
256 f << "ASCII\n";
257 f << "DATASET UNSTRUCTURED_GRID\n";
258
259 // ── Points ────────────────────────────────────────────────────────────────
260 f << "POINTS " << totalPts << " double\n";
261 for (Index k = 0; k < K; ++k)
262 {
263 const Real xL = mesh.leftVertex(k);
264 const Real h = mesh.elementSize(k);
265 for (Index i = 0; i < nFine; ++i)
266 {
267 const Real r = ref_pts[i];
268 const Real x = xL + (r + Real(1)) * Real(0.5) * h;
269 f << x << " 0.0 0.0\n";
270 }
271 }
272
273 // ── Cells ─────────────────────────────────────────────────────────────────
274 const Index cellListSize = K * (nFine + 1);
275 f << "\nCELLS " << K << " " << cellListSize << "\n";
276 for (Index k = 0; k < K; ++k)
277 {
278 f << nFine;
279 for (Index i = 0; i < nFine; ++i)
280 f << " " << k * nFine + i;
281 f << "\n";
282 }
283
284 f << "\nCELL_TYPES " << K << "\n";
285 for (Index k = 0; k < K; ++k)
286 f << "4\n";
287
288 // ── Point data ────────────────────────────────────────────────────────────
289 f << "\nPOINT_DATA " << totalPts << "\n";
290 f << "SCALARS " << fieldName << " double 1\n";
291 f << "LOOKUP_TABLE default\n";
292 for (Index k = 0; k < K; ++k)
293 {
294 const Real* uk = u.elementPtr(k);
295 const auto vals = solver.reconstruct(uk, ref_pts);
296 for (Index i = 0; i < nFine; ++i)
297 f << vals[i] << "\n";
298 }
299}
