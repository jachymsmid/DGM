93 // evaluate the Legendre polynomials at x
94 // Bonnet's formula see: https://proofwiki.org/wiki/Bonnet%27s_Recursion_Formula
101 static Real legendreP(Index n, Real x)
102 {
103 if ( n < 0 )
104 {
105 throw std::invalid_argument("The order of the Legendre polynomial must be non-negative ( n >= 0 )");
106 }
107 Real p0 = 1.0;
108 if (n == 0) return p0;
109 Real p1 = x;
110 if (n == 1) return p1;
111 Real norm, p2;
112 for (Index k = 1; k < n; ++k)
113 {
114 p2 = ((2 * k + 1) * x * p1 - k * p0) / (k + 1);
115 p0 = p1;
116 p1 = p2;
117 }
118 return p2;
119 }

121 // normalize the Legendre polynomial
122 // normalization n_n = \sqrt{(2n+1)/2}
129 static Real legendrePN(Index n, Real x)
130 {
131 Real norm = std::sqrt(Real(2*n+1) / 2);
132 return norm * legendreP(n,x);
133 }

135 // Evaluate the derivative of the Legendre derivative at x
136 // recursion formula, doesn't work for {-1, 1}
143 static Real legendrePDeriv(Index n, Real x)
144 {
145 if (n == 0) return Real(0);
146 if (n == 1) return Real(1);
147
148 if (x >= 1 - 1e-16)
149 {
150 return Real(n) * (Real(n) + 1) / 2.0;
151 }
152 else if (x <= -1 + 1e-16)
153 {
154 if (n % 2 == 0)
155 {
156 return -1.0 * Real(n) * (Real(n) + 1) / 2.0;
157 }
158 else
159 {
160 return Real(n) * (Real(n) + 1) / 2.0;
161 }
162 }
163
164 return n*(legendreP(n-1,x) - x*legendreP(n,x))/(1-x*x);
165 }
166
167 // normalize the derivative of the Legendre polynomial
168 // normalization n_n = \sqrt{(2n+1)/2}
175 static Real legendrePDerivN(Index n, Real x)
176 {
177 Real norm = std::sqrt(Real(2*n+1) / 2);
178 return norm * legendrePDeriv(n,x);
179 }
180
181 // Evalueate the second derivative of the Legendre polynomial
182 // recursion formula, doesn't work for {-1, 1}
183 // add source
190 static Real legendrePDeriv2(Index n, Real x)
191 {
192 if (std::abs(x) >= 1 - 1e-16)
193 {
194 throw std::invalid_argument("Division by zero in legendrePDeriv2()");
195 }
196
197 if (n < 2) return Real(0);
198 return (- n * ( n + 1 ) * legendreP(n,x) + 2 * x * legendrePDeriv(n,x))/(1 - x*x);
199 }
200
201 // evalueate the third derivative of the Legendre polynomial
202 // recursion formula, doesn't work for {-1, 1}
203 // add source
210 static Real legendrePDeriv3(Index n, Real x)
211 {
212 if (std::abs(x) >= 1 - 1e-16)
213 {
214 throw std::invalid_argument("Division by zero in legendrePDeriv3()");
215 }
216
217 if (n < 3) return Real(0);
218 return (4*x*legendrePDeriv2(n, x)-(n*(n+1)-2)*legendrePDeriv(n,x))/(1-x*x);
219 }

223 // Gauss–Lobatto–Legendre nodes: endpoints +-1 and interior zeros of dP_N(x)
224 // TODO: hardcode the points for low N
225 // @param N polynomial order of the approximation
226 void computeGLL_(Vector& r, Vector& w, const int N)
227 {
228 if (N < 1)
229 {
230 throw std::invalid_argument("Polynomial order must be at least 1");
231 }
232 else if ( r.getSize() != N + 1 || w.getSize() != N + 1)
233 {
234 throw std::invalid_argument("Invalid size of array r[] or w[] in function computeGLL_()");
235 }
236
237 // endpoints
238 r[0] = Real(-1);
239 r[N] = Real(+1);
240 w[0] = Real(2) / (N * (N + 1));
241 w[N] = w[0];
242
243 for (Index k = 1; k < N ; k++)
244 {
245 // Initial guess: Chebyshev-like
246 constexpr Real pi = std::acos(Real(-1));
247 Real x = - std::cos(pi * k / N);
248 for (int iter = 0; iter < 30; iter++)
249 {
250 Real dPn = legendrePDeriv(N, x);
251 Real ddPn = legendrePDeriv2(N, x);
252 Real dx = dPn / ddPn;
253 x -= dx;
254 if (std::abs(dx) < 1e-15) break;
255 }
256
257 r[k] = x;
258 Real Pn = legendreP(N, x);
259 w[k] = Real(2) / (N * (N + 1) * Pn * Pn);
260 }
261 }

263 // Vandermonde matrix V_{ij} = sqrt((2j+1)/2) * P_j(r_i)
264 // (orthonormal Legendre basis, as in Hesthaven eq. 3.1)
265 Matrix buildVandermonde_(const Vector& r) const
266 {
267 Matrix V(Np_, Np_);
268 for (Index i = 0; i < Np_; i++)
269 {
270 for (Index j = 0; j < Np_; j++)
271 {
272 V(i,j) = legendrePN(j, r[i]);
273 }
274 }
275 return V;
276 }

278 // derivative of the Vandermonde matrix (normalized)
279 // dV_{ij} = sqrt((2j+1)/2) * P'_j(r_i)
280 Matrix buildDerivVandermonde_(const Vector& r) const
281 {
282 Matrix Vr(Np_, Np_);
283 for (Index i = 0; i < Np_; ++i)
284 {
285 for (Index j = 0; j < Np_; ++j)
286 {
287 Vr(i,j) = legendrePDerivN(j, r[i]);
288 }
289 }
290 return Vr;
291 }

293 // Convert TNL DenseMatrix to Eigen MatrixXd
294 Eigen::MatrixXd tnlToEigen(const Matrix& A) const
295 {
296 int rows = A.getRows();
297 int cols = A.getColumns();
298 Eigen::MatrixXd E(rows, cols);
299 for (int i = 0; i < rows; i++)
300 {
301 for (int j = 0; j < cols; j++)
302 {
303 E(i, j) = A.getElement(i, j);
304 }
305 }
306 return E;
307 }

309 // Convert Eigen MatrixXd to TNL DenseMatrix
310 Matrix eigenToTnl(const Eigen::MatrixXd& E) const
311 {
312 Matrix A;
313 A.setDimensions(E.rows(), E.cols()); // must come before any setElement
314 for (int i = 0; i < E.rows(); i++)
315 {
316 for (int j = 0; j < E.cols(); j++)
317 {
318 A.setElement(i, j, E(i, j));
319 }
320 }
321 return A;
322 }

324 // Compute the inverse of a square DenseMatrix using Eigen
325 Matrix invertMatrix_(const Matrix& A) const
326 {
327 Eigen::MatrixXd E = tnlToEigen(A);
328 E = E.inverse();
329 return eigenToTnl(E);

332 // build the differentiation matrix for the weak formulation
333 // D = V Vr^T M
334 Matrix buildDwMatrix_(const Matrix& V, const Matrix& Vr, const Matrix& M) const
335 {
336 Matrix tmp, tmp2, VrT;
337 VrT.getTransposition(Vr);
338 tmp.getMatrixProduct(V, VrT);
339 tmp2.getMatrixProduct(tmp, M);
340
341 return tmp2;
342 }

344 // build the differentiation matrix for the strong formulation
345 // D = Vr V^(-1)
346 Matrix buildDMatrix_(const Matrix& V, const Matrix& Vr) const
347 {
348 Matrix tmp, V_inv;
349 V_inv.setDimensions(Np_, Np_);
350 V_inv = invertMatrix_(V);
351 tmp.getMatrixProduct(Vr, V_inv);
352
353 return tmp;
354 }

356 // M = (V V^T)^{-1}
357 Matrix massMatrix_(const Matrix& V) const
358 {
359 Matrix VT, tmp;
360 VT.getTransposition(V);
361 tmp.getMatrixProduct(V, VT);
362
363 return invertMatrix_(tmp);
364 }

366 // build the LIFT matrix
367 Matrix buildLIFT_(const Matrix& V) const
368 {
369 Matrix tmp, VT;
370 VT.getTransposition(V);
371 tmp.getMatrixProduct(V, VT);
372
373 Matrix L(Np_, 2);
374 for (Index i = 0; i < Np_; i++) {
375 L(i,0) = tmp(i, 0);
376 L(i,1) = tmp(i, Np_-1);
377 }
378 return L;
379 }

392 Real conditionNumber_(const Matrix& matrix) const
393 {
394 const Eigen::MatrixXd eigenMatrix = tnlToEigen(matrix);
395 Eigen::JacobiSVD<Eigen::MatrixXd> svd(eigenMatrix, Eigen::ComputeThinU | Eigen::ComputeThinV);
396 const auto& singularValues = svd.singularValues();
397
398 if (singularValues.size() == 0)
399 {
400 return Real(0);
401 }
402
403 const Real maxSingular = singularValues.maxCoeff();
404 const Real minSingular = singularValues.minCoeff();
405
406 if (minSingular <= std::numeric_limits<Real>::epsilon())
407 {
408 return std::numeric_limits<Real>::infinity();
409 }
410 return maxSingular / minSingular;
411 }
