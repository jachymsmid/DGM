16// ---------------------- Base numerical flux class ---------------------------
17// Abstract base (crtp-free for simplicity; use std::function or templates)
18template< class Real >
26struct NumericalFlux
27{
28 virtual ~NumericalFlux() = default;
29 // f*: numerical flux at interface given the interior state u_minus (value
30 // from the element on the "left" side of the interface), the exterior state
31 // u_plus (right side), and the outward normal from the left element
32 // (n_outward == -1 for the left face, +1 for the right face). The function
33 // returns the scalar numerical flux f* (already oriented along the 1D
34 // coordinate) to be used directly in the surface term assembly.
35 virtual Real compute(Real u_minus, Real u_plus, Real n_outward) const = 0;
36};

38// -------------------------------- Upwind ------------------------------------
39// f*(u) = f(u-)
40template< class Real >
47struct UpwindFlux : NumericalFlux< Real >
48{
49 // constructor
50 explicit UpwindFlux(std::function< Real( Real u ) > advection_speed, std::function< Real( Real u ) > physical_flux) : advection_speed_(std::move(advection_speed)), physical_flux_(std::move(physical_flux)) {}
51
52 Real compute(Real u_int, Real u_ext, Real n_outward) const override
53 {
54 // u_minus is the interior (left) state, u_plus is the exterior (right)
55 // If a * n_outward >= 0 the characteristic leaves the interior cell so
56 // the interior state determines the flux; otherwise use the exterior.
57 Real C = advection_speed_(u_int);
58 if (C * n_outward >= Real(0))
59 return physical_flux_(u_int);
60 else
61 return physical_flux_(u_ext);
62 }
63
64 // data members
65 std::function< Real( Real u ) > advection_speed_;
66 std::function< Real( Real u ) > physical_flux_;
67};

70// ----------------------------- Lax-Friedrichs -------------------------------
71// f* = 0.5*(f(u-) + f(u+)) - 0.5*C*(u+ - u-)
72template< class Real >
80struct LaxFriedrichsFlux : NumericalFlux< Real >
81{
82 explicit LaxFriedrichsFlux(
83 std::function< Real( Real u ) > advection_speed,
84 std::function< Real( Real u ) > physical_flux ) :
85 advection_speed_(std::move(advection_speed)),
86 physical_flux_(std::move(physical_flux)) {}
87
88 Real compute(Real u_minus, Real u_plus, Real n_outward) const override
89 {
90 // C is an estimate of the maximum wave speed across the interface
91 Real C = TNL::argAbsMax(advection_speed_(u_minus), advection_speed_(u_plus));
92 return Real(0.5) * (physical_flux_(u_minus) + physical_flux_(u_plus))
93 + Real(0.5) * C * n_outward * (u_minus - u_plus);
94 }
95
96 std::function< Real( Real u ) > advection_speed_;
97 std::function< Real( Real u ) > physical_flux_;
98};

110
111 Real compute(Real u_minus, Real u_plus, Real n_outward) const override

115 return TNL::min(physical_flux_(u_minus), physical_flux_(u_plus));
