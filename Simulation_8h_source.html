215 void
216 computeLocalTimeSteps( const RealType CFL )
217 {
218 const auto innerCellDTVectors_const_view = meshData.primary.innerCellDTVectors.getConstView();
219 const auto innerCellIDs_const_view = meshData.IDs.innerCellIDs.getConstView();
220 const auto w_old_const_view = fluidData.w_old.getConstView();
221 auto innerCellDTs_view = fluidData.innerCellDTs.getView();
222
223 auto computeSingleTimeStep = [ = ] __cuda_callable__( const GlobalIndexType iterIndex ) mutable
224 {
225 const GlobalIndexType cellIndex = innerCellIDs_const_view[ iterIndex ];
226
227 const Primitive< Traits > currPV( w_old_const_view[ cellIndex ] );
228 const RealType u = currPV.u;
229 const RealType v = currPV.v;
230 const RealType c = currPV.c;
231
232 const VectorType xi = innerCellDTVectors_const_view( iterIndex, 0 );
233 const VectorType eta = innerCellDTVectors_const_view( iterIndex, 1 );
234
235 const RealType xi_norm = TNL::l2Norm( xi );
236 const RealType eta_norm = TNL::l2Norm( eta );
237
238 const RealType u_xi = fabs( ( u * xi.x() + v * xi.y() ) / xi_norm );
239 const RealType u_eta = fabs( ( u * eta.x() + v * eta.y() ) / eta_norm );
240
241 const RealType d_xi = ( u_xi + c ) / xi_norm;
242 const RealType d_eta = ( u_eta + c ) / eta_norm;
243
244 const RealType res = CFL / ( d_xi + d_eta );
245 innerCellDTs_view[ iterIndex ] = res;
246 };
247 parallelFor< Device >( 0, meshData.IDs.innerCellIDs.getSize(), computeSingleTimeStep );
248 }

306 {

308 const auto innerCellNeighborOffsets_const_view = meshData.connectivity.innerCellNeighborOffsets.getConstView();

311 const auto cellNeighborOffsets_const_view = meshData.connectivity.cellNeighborOffsets.getConstView();
312 const auto cellNeighborCounts_const_view = meshData.connectivity.cellNeighborCounts.getConstView();
313 const auto cellNeighborIDs_const_view = meshData.connectivity.cellNeighborIDs.getConstView();
314
315 const auto innerCellScaledNormals_const_view = meshData.primary.innerCellScaledNormals.getConstView();
316 const auto dualMeshScaledNormals_const_view = meshData.dual.dualMeshScaledNormals.getConstView();
317 const auto dualMeshVolumes_const_view = meshData.dual.dualMeshVolumes.getConstView();
318 const auto innerCellIDs_const_view = meshData.IDs.innerCellIDs.getConstView();
319
320 const auto wi_reconstructed_const_view = fluidData.wi_reconstructed.getConstView();
321 const auto wj_reconstructed_const_view = fluidData.wj_reconstructed.getConstView();
322 const auto w_points_const_view = fluidData.w_points.getConstView();
323 const auto w_old_const_view = fluidData.w_old.getConstView();
324 auto innerCellR_view = fluidData.innerCellR.getView();
325
326 auto computeFlux = [ = ] __cuda_callable__( const GlobalIndexType iterIndex ) mutable
327 {
328 const auto cellIndex = innerCellIDs_const_view[ iterIndex ];
329
330 ConservativeType convectiveTermsSum{};
331 ConservativeType diffusiveTermsSum{};
332 const auto innerOffset = innerCellNeighborOffsets_const_view[ iterIndex ];
333 const auto globalOffset = cellNeighborOffsets_const_view[ cellIndex ];
334 const auto neighborCount = innerCellNeighborCounts_const_view[ iterIndex ];
335 const auto wi = w_old_const_view[ cellIndex ];
336 for( LocalIndexType neighborIndex = 0; neighborIndex < neighborCount; ++neighborIndex ) {
337 const auto scaledNormal = innerCellScaledNormals_const_view[ innerOffset + neighborIndex ];
338 const auto wi_reconstructed = wi_reconstructed_const_view[ globalOffset + neighborIndex ];
339 const auto wj_reconstructed = wj_reconstructed_const_view[ globalOffset + neighborIndex ];
340 const auto convectiveIncrement = HLLC< Traits >( wi_reconstructed, wj_reconstructed, scaledNormal );
341 convectiveTermsSum += convectiveIncrement;
342
343 const auto cellNeighborIndex = innerCellNeighborIDs_const_view[ innerOffset + neighborIndex ];
344 const auto vertexIndexR = innerCellInterfacePoints_const_view( innerOffset + neighborIndex, 0 );
345 const auto vertexIndexL = innerCellInterfacePoints_const_view( innerOffset + neighborIndex, 1 );
346 const auto wj = w_old_const_view[ cellNeighborIndex ];
347 const auto wr = w_points_const_view[ vertexIndexR ];
348 const auto wl = w_points_const_view[ vertexIndexL ];
349 const auto sn1 = dualMeshScaledNormals_const_view( innerOffset + neighborIndex, 0 );
350 const auto sn2 = dualMeshScaledNormals_const_view( innerOffset + neighborIndex, 1 );
351 const auto sn3 = dualMeshScaledNormals_const_view( innerOffset + neighborIndex, 2 );
352 const auto sn4 = dualMeshScaledNormals_const_view( innerOffset + neighborIndex, 3 );
353 const auto dualMeshVolume = dualMeshVolumes_const_view[ innerOffset + neighborIndex ];
354 const auto diffusiveIncrement =
355 computeDiffusiveTerms( wi, wj, wr, wl, scaledNormal, sn1, sn2, sn3, sn4, dualMeshVolume );
356 diffusiveTermsSum += diffusiveIncrement;
357 }
358 innerCellR_view[ iterIndex ] = convectiveTermsSum - diffusiveTermsSum;
359 };
360 parallelFor< Device >( 0, meshData.IDs.innerCellIDs.getSize(), computeFlux );
361 }

387 RealType
388 computeRezi()
389 {
390 const auto innerCellIDs_const_view = meshData.IDs.innerCellIDs.getConstView();
391 const auto innerCellAreas_const_view = meshData.primary.innerCellAreas.getConstView();
392 const auto innerCellDTs_const_view = fluidData.innerCellDTs.getConstView();
393 const auto w_old_const_view = fluidData.w_old.getConstView();
394 const auto w_new_const_view = fluidData.w_new.getConstView();
395
396 // get value from one cell
397 auto fetch = [ = ] __cuda_callable__( const GlobalIndexType iterIndex )
398 {
399 const GlobalIndexType cellIndex = innerCellIDs_const_view[ iterIndex ];
400 const RealType cellArea = innerCellAreas_const_view[ iterIndex ];
401 const RealType cellDT = innerCellDTs_const_view[ iterIndex ];
402 const RealType rho_new = w_new_const_view[ cellIndex ].getElement( 0 );
403 const RealType rho_old = w_old_const_view[ cellIndex ].getElement( 0 );
404
405 return cellArea * pow( ( rho_new - rho_old ) / cellDT, 2 );
406 };
407
408 auto reduction = [ = ] __cuda_callable__( const RealType& sum, const RealType& increment )
409 {
410 return sum + increment;
411 };
412
413 // parallel reduction
414 return log( sqrt( reduce< Device >( 0, meshData.IDs.innerCellIDs.getSize(), fetch, reduction, 0.0 ) ) );
415 }
